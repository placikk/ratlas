# Plan 3: Mapa szczurów — implementacja

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Zakładka „Mapa” w Ratlasie pokazuje zgłoszenia z ostatnich 7 dni jako zanikające pinezki-szczury (z „gniazdami” przy oddaleniu), panel ze zdjęciem po kliknięciu i zgłaszanie nadużyć; Profil dostaje listę „Moje zgłoszenia” ze zdjęciami i statusami.

**Architecture:** Dane mapy daje funkcja SQL `sightings_in_view(bbox)` (`security definer`, tylko dla `authenticated`), zgłoszenia nadużyć — `report_sighting` (ukrywa pinezkę po 3 zgłoszeniach). Mapa to MapLibre GL 6 z kafelkami OpenFreeMap (styl Positron przekolorowany w kodzie), ładowana leniwie (`React.lazy`), żeby nie powiększać startu aplikacji. Logika (zanikanie, GeoJSON, „X temu”, przekolorowanie stylu) jest w czystych modułach testowanych Vitestem; komponent `MapView` (WebGL) jest cienki i w testach podmieniany.

**Tech Stack:** jak w planach 1–2 + `maplibre-gl@^6` (ESM; worker przez `?worker&url` + `setWorkerUrl` — sprawdzone w headless Chromium), OpenFreeMap (`https://tiles.openfreemap.org/styles/positron`).

**Spec:** [docs/superpowers/specs/2026-09-29-szczury-wroclaw-design.md](../specs/2026-09-29-szczury-wroclaw-design.md) — sekcje 4 (zgłaszanie nadużyć), 5 (`reports`), 7 (mapa), 11 (prywatność)

**Poprzednie plany:** [plan 1](2026-09-30-plan-1-fundament.md), [plan 2](2026-10-01-plan-2-zgloszenia.md) — zrealizowane; kod w repozytorium jest źródłem prawdy.

```sql
-- Mapa: publiczne pinezki i zgłaszanie nadużyć.
-- Produkcja ma wyłączone automatyczne udostępnianie — uprawnienia nadajemy jawnie.

create function public.sightings_in_view(
  p_west double precision,
  p_south double precision,
  p_east double precision,
  p_north double precision
)
returns table (
  id uuid,
  lon double precision,
  lat double precision,
  rats integer,
  created_at timestamptz,
  photo_path text,
  nickname text,
  is_mine boolean
)
language sql
stable
security definer
set search_path = ''
as $$
  select s.id,
         extensions.st_x(s.location::extensions.geometry),
         extensions.st_y(s.location::extensions.geometry),
         least(coalesce(s.rats_detected, 0), public.config_int('max_rats_per_photo')),
         s.created_at,
         s.photo_path,
         case when p.show_nickname_on_map then p.nickname end,
         s.user_id = (select auth.uid())
    from public.sightings s
    join public.profiles p on p.id = s.user_id
   where s.status in ('accepted', 'accepted_no_points')
     and s.public_on_map
     and not s.hidden
     and s.created_at > now() - make_interval(days => public.config_int('pin_fade_days'))
     and extensions.st_intersects(
           s.location,
           extensions.st_makeenvelope(p_west, p_south, p_east, p_north, 4326)::extensions.geography
         )
   order by s.created_at desc
   limit 500
$$;

revoke execute on function public.sightings_in_view(double precision, double precision, double precision, double precision) from public, anon, authenticated;
grant execute on function public.sightings_in_view(double precision, double precision, double precision, double precision) to authenticated;

create table public.reports (
  id bigint generated always as identity primary key,
  sighting_id uuid not null references public.sightings (id) on delete cascade,
  reporter_id uuid not null references public.profiles (id) on delete cascade,
  reason text not null check (reason in ('not_rat', 'inappropriate', 'other')),
  created_at timestamptz not null default now(),
  unique (sighting_id, reporter_id)
);

alter table public.reports enable row level security;
revoke all on public.reports from anon, authenticated;

create function public.report_sighting(p_sighting_id uuid, p_reason text)
returns text
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_user uuid := auth.uid();
  v_owner uuid;
  v_reports integer;
begin
  if v_user is null then
    raise exception 'not authenticated' using errcode = '28000';
  end if;

  -- Zgłosić można tylko to, co widać na mapie.
  select s.user_id into v_owner
    from public.sightings s
   where s.id = p_sighting_id
     and s.public_on_map
     and not s.hidden
     and s.status in ('accepted', 'accepted_no_points')
   for update;

  if v_owner is null then
    return 'not_found';
  end if;
  if v_owner = v_user then
    return 'own';
  end if;

  insert into public.reports (sighting_id, reporter_id, reason)
  values (p_sighting_id, v_user, p_reason)
  on conflict (sighting_id, reporter_id) do nothing;
  if not found then
    return 'already';
  end if;

  select count(*) into v_reports from public.reports where sighting_id = p_sighting_id;
  if v_reports >= public.config_int('reports_to_hide') then
    update public.sightings set hidden = true where id = p_sighting_id;
    return 'hidden';
  end if;

  return 'reported';
end;
$$;

revoke execute on function public.report_sighting(uuid, text) from public, anon, authenticated;
grant execute on function public.report_sighting(uuid, text) to authenticated;
```

```ts
export interface Pin {
  id: string
  lon: number
  lat: number
  rats: number
  createdAt: string
  photoPath: string
  nickname: string | null
  isMine: boolean
}

export interface Bounds {
  west: number
  south: number
  east: number
  north: number
}

export interface PinProperties {
  id: string
  rats: number
  opacity: number
  mine: boolean
}

export interface PinFeature {
  type: 'Feature'
  id: number
  properties: PinProperties
  geometry: { type: 'Point'; coordinates: [number, number] }
}

export interface PinCollection {
  type: 'FeatureCollection'
  features: PinFeature[]
}

export const DEFAULT_FADE_DAYS = 7
export const MIN_PIN_OPACITY = 0.15

const DAY_MS = 86_400_000

// Świeża pinezka jest w pełni widoczna, a z wiekiem blednie aż do MIN_PIN_OPACITY.
export function pinOpacity(createdAt: string, now: number, fadeDays: number): number {
  const ageDays = (now - Date.parse(createdAt)) / DAY_MS
  return Math.max(MIN_PIN_OPACITY, Math.min(1, 1 - ageDays / fadeDays))
}

export function toFeatureCollection(pins: Pin[], now: number, fadeDays: number): PinCollection {
  return {
    type: 'FeatureCollection',
    features: pins.map((pin, index) => ({
      type: 'Feature',
      id: index,
      properties: { id: pin.id, rats: pin.rats, opacity: pinOpacity(pin.createdAt, now, fadeDays), mine: pin.isMine },
      geometry: { type: 'Point', coordinates: [pin.lon, pin.lat] },
    })),
  }
}

const UNITS: [Intl.RelativeTimeFormatUnit, number][] = [
  ['day', DAY_MS],
  ['hour', 3_600_000],
  ['minute', 60_000],
]

export function timeAgo(createdAt: string, now: number, justNow: string): string {
  const diff = Date.parse(createdAt) - now
  const format = new Intl.RelativeTimeFormat('pl', { numeric: 'auto' })
  for (const [unit, size] of UNITS) {
    if (Math.abs(diff) >= size) return format.format(Math.round(diff / size), unit)
  }
  return justNow
}
```

```ts
import { supabase } from '../lib/supabase'
import { DEFAULT_FADE_DAYS, type Bounds, type Pin } from './pins'

export type ReportReason = 'not_rat' | 'inappropriate' | 'other'
export const REPORT_REASONS: ReportReason[] = ['not_rat', 'inappropriate', 'other']
export type ReportResult = 'reported' | 'already' | 'own' | 'hidden' | 'not_found' | 'error'

const REPORT_RESULTS: readonly string[] = ['reported', 'already', 'own', 'hidden', 'not_found']

interface PinRow {
  id: string
  lon: number
  lat: number
  rats: number
  created_at: string
  photo_path: string
  nickname: string | null
  is_mine: boolean
}

export async function fetchPins(bounds: Bounds): Promise<Pin[]> {
  const { data, error } = await supabase.rpc('sightings_in_view', {
    p_west: bounds.west,
    p_south: bounds.south,
    p_east: bounds.east,
    p_north: bounds.north,
  })
  if (error) throw error
  return ((data ?? []) as PinRow[]).map((row) => ({
    id: row.id,
    lon: row.lon,
    lat: row.lat,
    rats: row.rats,
    createdAt: row.created_at,
    photoPath: row.photo_path,
    nickname: row.nickname,
    isMine: row.is_mine,
  }))
}

export async function fetchFadeDays(): Promise<number> {
  const { data, error } = await supabase.from('app_config').select('value').eq('key', 'pin_fade_days').maybeSingle()
  const days = Number((data as { value?: unknown } | null)?.value)
  return error || !Number.isFinite(days) || days <= 0 ? DEFAULT_FADE_DAYS : days
}

export async function reportPin(sightingId: string, reason: ReportReason): Promise<ReportResult> {
  const { data, error } = await supabase.rpc('report_sighting', { p_sighting_id: sightingId, p_reason: reason })
  if (error || typeof data !== 'string' || !REPORT_RESULTS.includes(data)) return 'error'
  return data as ReportResult
}

export function photoUrl(path: string): string {
  return supabase.storage.from('map-photos').getPublicUrl(path).data.publicUrl
}
```

```ts
import type { LayerSpecification, StyleSpecification } from 'maplibre-gl'

export const BASE_STYLE_URL = 'https://tiles.openfreemap.org/styles/positron'

// Kolory warstw podkładu (Positron z OpenFreeMap) dopasowane do palety Ratlasa.
const RECOLOR: Record<string, Record<string, string>> = {
  background: { 'background-color': '#F4F1EF' },
  road_area_pier: { 'fill-color': '#F4F1EF' },
  water: { 'fill-color': '#D3CCCD' },
  building: { 'fill-color': '#EBE5E2' },
  landuse_residential: { 'fill-color': '#EFEAE7' },
}

export function ratifyStyle(style: StyleSpecification): StyleSpecification {
  return {
    ...style,
    layers: style.layers.map((layer) => {
      const colors = RECOLOR[layer.id]
      if (!colors) return layer
      return { ...layer, paint: { ...(layer as { paint?: object }).paint, ...colors } } as LayerSpecification
    }),
  }
}
```

```tsx
import { useCallback, useEffect, useMemo, useRef, useState } from 'react'
import { useTranslation } from 'react-i18next'
import { fetchFadeDays, fetchPins } from './mapApi'
import { MapView, type MapLabels } from './MapView'
import { DEFAULT_FADE_DAYS, toFeatureCollection, type Bounds, type Pin } from './pins'
import { PinPanel } from './PinPanel'

export const MAP_REFRESH_MS = 60_000

export function MapScreen() {
  const { t } = useTranslation()
  const [pins, setPins] = useState<Pin[]>([])
  const [fadeDays, setFadeDays] = useState(DEFAULT_FADE_DAYS)
  const [now, setNow] = useState(() => Date.now())
  const [failed, setFailed] = useState(false)
  const [selectedId, setSelectedId] = useState<string | null>(null)
  const boundsRef = useRef<Bounds | null>(null)
  const requestRef = useRef(0)

  // Odpowiedź spóźnionego zapytania (np. po szybkim przesuwaniu mapy) jest pomijana.
  const load = useCallback(async (bounds: Bounds) => {
    boundsRef.current = bounds
    const request = ++requestRef.current
    try {
      const loaded = await fetchPins(bounds)
      if (request !== requestRef.current) return
      setPins(loaded)
      setFailed(false)
    } catch {
      if (request === requestRef.current) setFailed(true)
    }
    setNow(Date.now())
  }, [])

  useEffect(() => {
    let active = true
    void fetchFadeDays().then((days) => {
      if (active) setFadeDays(days)
    })
    return () => {
      active = false
    }
  }, [])

  useEffect(() => {
    const timer = setInterval(() => {
      if (boundsRef.current) void load(boundsRef.current)
    }, MAP_REFRESH_MS)
    return () => clearInterval(timer)
  }, [load])

  const collection = useMemo(() => toFeatureCollection(pins, now, fadeDays), [pins, now, fadeDays])
  const selected = pins.find((pin) => pin.id === selectedId) ?? null

  const labels: MapLabels = {
    zoomIn: t('map.zoomIn'),
    zoomOut: t('map.zoomOut'),
    locate: t('map.locate'),
    locationUnavailable: t('map.locationUnavailable'),
    attribution: t('map.attribution'),
  }

  return (
    <section className="map-screen">
      <MapView collection={collection} labels={labels} onSelect={setSelectedId} onBoundsChange={(bounds) => void load(bounds)} />
      {failed && (
        <p className="map-screen__error" role="alert">
          {t('map.loadFailed')}
        </p>
      )}
      {selected && (
        <PinPanel
          key={selected.id}
          pin={selected}
          now={now}
          onClose={() => setSelectedId(null)}
          onHidden={(id) => setPins((current) => current.filter((pin) => pin.id !== id))}
        />
      )}
    </section>
  )
}
```

```ts
import { supabase } from '../lib/supabase'
import type { RecordedStatus } from '../report/reportApi'

export interface MySighting {
  id: string
  status: RecordedStatus
  rejectReason: string | null
  ratsAwarded: number
  tokensAwarded: number
  createdAt: string
  photoUrl: string | null
}

interface SightingRow {
  id: string
  status: RecordedStatus
  reject_reason: string | null
  rats_awarded: number
  tokens_awarded: number
  created_at: string
  photo_bucket: string
  photo_path: string
}

const PUBLIC_BUCKET = 'map-photos'
const PRIVATE_BUCKET = 'sightings-private'
const SIGNED_URL_SECONDS = 3600

export async function fetchMySightings(limit = 10): Promise<MySighting[]> {
  const { data, error } = await supabase
    .from('sightings')
    .select('id, status, reject_reason, rats_awarded, tokens_awarded, created_at, photo_bucket, photo_path')
    .order('created_at', { ascending: false })
    .limit(limit)
  if (error) throw error
  const rows = (data ?? []) as SightingRow[]

  // Prywatne zdjęcia (selfie, odrzucone, oczekujące) widzi tylko autor — przez podpisany, czasowy adres.
  const privatePaths = rows.filter((row) => row.photo_bucket !== PUBLIC_BUCKET).map((row) => row.photo_path)
  const signedByPath = new Map<string, string>()
  if (privatePaths.length > 0) {
    const signed = await supabase.storage.from(PRIVATE_BUCKET).createSignedUrls(privatePaths, SIGNED_URL_SECONDS)
    for (const entry of signed.data ?? []) {
      if (entry.path && entry.signedUrl) signedByPath.set(entry.path, entry.signedUrl)
    }
  }

  return rows.map((row) => ({
    id: row.id,
    status: row.status,
    rejectReason: row.reject_reason,
    ratsAwarded: row.rats_awarded,
    tokensAwarded: row.tokens_awarded,
    createdAt: row.created_at,
    photoUrl:
      row.photo_bucket === PUBLIC_BUCKET
        ? supabase.storage.from(PUBLIC_BUCKET).getPublicUrl(row.photo_path).data.publicUrl
        : signedByPath.get(row.photo_path) ?? null,
  }))
}
```

```tsx
import { useEffect, useState } from 'react'
import { useTranslation } from 'react-i18next'
import { timeAgo } from '../map/pins'
import { fetchMySightings, type MySighting } from './mySightingsApi'

type State = { kind: 'loading' } | { kind: 'failed' } | { kind: 'loaded'; items: MySighting[] }

export function MySightings() {
  const { t } = useTranslation()
  const [state, setState] = useState<State>({ kind: 'loading' })

  useEffect(() => {
    let active = true
    fetchMySightings().then(
      (items) => {
        if (active) setState({ kind: 'loaded', items })
      },
      () => {
        if (active) setState({ kind: 'failed' })
      },
    )
    return () => {
      active = false
    }
  }, [])

  function statusText(item: MySighting): string {
    const status = t(`mySightings.status.${item.status}`)
    return item.status === 'rejected' && item.rejectReason
      ? `${status}: ${t(`mySightings.reasons.${item.rejectReason}`)}`
      : status
  }

  return (
    <section className="stack">
      <h2>{t('mySightings.title')}</h2>
      {state.kind === 'loading' && <p className="muted">{t('app.loading')}</p>}
      {state.kind === 'failed' && (
        <p className="error" role="alert">
          {t('mySightings.loadFailed')}
        </p>
      )}
      {state.kind === 'loaded' && state.items.length === 0 && <p className="muted">{t('mySightings.empty')}</p>}
      {state.kind === 'loaded' && state.items.length > 0 && (
        <ul className="my-sightings">
          {state.items.map((item) => (
            <li key={item.id} className="card my-sightings__item">
              {item.photoUrl ? (
                <img className="my-sightings__photo" src={item.photoUrl} alt={t('mySightings.photoAlt')} loading="lazy" />
              ) : (
                <span className="my-sightings__photo" aria-hidden="true">
                  🐀
                </span>
              )}
              <div className="stack my-sightings__text">
                <strong>{statusText(item)}</strong>
                {item.status === 'accepted' && (
                  <span>
                    {t('mySightings.gain', {
                      rats: t('result.rats', { count: item.ratsAwarded }),
                      raciki: t('result.raciki', { count: item.tokensAwarded }),
                    })}
                  </span>
                )}
                <span className="muted">{timeAgo(item.createdAt, Date.now(), t('map.justNow'))}</span>
              </div>
            </li>
          ))}
        </ul>
      )}
    </section>
  )
}
```

````markdown
```sql
begin;
update public.profiles p
   set total_rats = greatest(0, p.total_rats - s.rats_awarded),
       tokens_balance = greatest(0, p.tokens_balance - s.tokens_awarded)
  from public.sightings s
 where s.id = '<id zgłoszenia>' and p.id = s.user_id;
insert into public.token_ledger (user_id, delta, reason, ref_id)
select user_id, -tokens_awarded, 'correction', id from public.sightings where id = '<id zgłoszenia>';
commit;
```
````
